#############################################
SimOpInt Class Python Console Commands
#############################################

=========================
Thread Commands
=========================

* Create Interface thread (from simopintd)
simopintd.getInterface().simopint_thread = threading.Thread(target=self.mainLoop)

* Start Thread
simopintd.getInterface().simopint_thread.start()

=========================
Status Commands
=========================

* Get Interface status
simopintd.getInterface().getStatus()

* Set simopint status [ statusid : 0 => interface stopped, 1 => interface created / interface loop stopped , 2 => interface started / interface loop started ]
simopintd.getInterface().setStatus(<statusid) # statusid is int

=========================
Interface Commands
=========================

* Start Server
simopintd.getInterface().startInterface()

* Stop Server
simopintd.getInterface().stopInterface()

=========================
Interface Main Loop Commands
=========================

* Start interface main loop
simopintd.getInterface().startIntLoop()

* Stop interface main loop
simopintd.getInterface().stopIntLoop()
